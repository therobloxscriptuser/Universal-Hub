local function protectGui(gui)
    if gethui then
        gui.Parent = gethui()
    elseif syn and syn.protect_gui then
        syn.protect_gui(gui)
        gui.Parent = CoreGui
    else
        local ok = pcall(function() gui.Parent = CoreGui end)
        if not ok then
            gui.Parent = Players.LocalPlayer:WaitForChild("PlayerGui")
        end
    end
end

local function create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        inst[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    return inst
end

local function corner(radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius) })
end

local function stroke(color, thickness, transparency)
    return create("UIStroke", {
        Color = color,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
end

local function tween(inst, time, props)
    local t = TweenService:Create(inst, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local function trim(s)
    return (s:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function makeButton(name, text, xScale, xOffset, color)
    local btn = create("TextButton", {
        Name = name,
        Position = UDim2.new(xScale, xOffset, 0, 162),
        Size = UDim2.new(0.5, -26, 0, 42),
        BackgroundColor3 = color,
        Text = text,
        TextColor3 = Theme.Text,
        Font = Enum.Font.GothamBold,
        TextSize = 15,
        AutoButtonColor = false,
        Parent = Main,
    }, { corner(10) })

    local base = color
    local hover = Color3.new(
        math.min(base.R + 0.06, 1),
        math.min(base.G + 0.06, 1),
        math.min(base.B + 0.06, 1)
    )
    btn.MouseEnter:Connect(function() tween(btn, 0.15, { BackgroundColor3 = hover }) end)
    btn.MouseLeave:Connect(function() tween(btn, 0.15, { BackgroundColor3 = base }) end)
    return btn
end

local busy = false

local function setStatus(text, color)
    Status.Text = text
    Status.TextColor3 = color or Theme.SubText
end

local function shake()
    local origin = Main.Position
    for _, offset in ipairs({ 8, -8, 5, -5, 0 }) do
        tween(Main, 0.04, { Position = origin + UDim2.new(0, offset, 0, 0) })
        task.wait(0.04)
    end
end

local function validateKey(key)
    for _, valid in ipairs(Config.ValidKeys) do
        if key == valid then
            return true
        end
    end

    if Config.ApiUrl ~= "" then
        local ok, response = pcall(function()
            return game:HttpGet(Config.ApiUrl .. key)
        end)
        if ok and type(response) == "string" and trim(response):lower() == "true" then
            return true
        end
    end

    return false
end

local function closeGui()
    tween(Main, 0.25, { Size = UDim2.new(0, 0, 0, 0) })
    task.wait(0.3)
    ScreenGui:Destroy()
end

CheckBtn.MouseButton1Click:Connect(function()
    if busy then return end
    local key = trim(KeyBox.Text)

    if key == "" then
        setStatus("Please enter a key first.", Theme.Error)
        tween(InputStroke, 0.2, { Color = Theme.Error })
        task.spawn(shake)
        return
    end

    busy = true
    CheckBtn.Text = "Checking..."
    setStatus("Verifying key...", Theme.SubText)
    tween(InputStroke, 0.2, { Color = Theme.Accent })

    local valid = validateKey(key)

    if valid then
        CheckBtn.Text = "Success!"
        setStatus("Key accepted. Loading Universal HUB...", Theme.Success)
        tween(InputStroke, 0.2, { Color = Theme.Success })
        task.wait(1)
        closeGui()
        task.spawn(Config.OnSuccess)
    else
        CheckBtn.Text = "Check Key"
        setStatus("Invalid key. Please try again.", Theme.Error)
        tween(InputStroke, 0.2, { Color = Theme.Error })
        task.spawn(shake)
        busy = false
    end
end)

CopyBtn.MouseButton1Click:Connect(function()
    if Config.KeyLink == "" or Config.KeyLink == "PASTE_YOUR_KEY_LINK_HERE" then
        setStatus("No key link has been set yet.", Theme.Error)
        return
    end

    local copied = false
    if setclipboard then
        copied = pcall(setclipboard, Config.KeyLink)
    elseif toclipboard then
        copied = pcall(toclipboard, Config.KeyLink)
    end

    if copied then
        CopyBtn.Text = "Copied!"
        setStatus("Link copied. Open it in your browser to get a key.", Theme.Success)
        task.delay(1.5, function()
            if CopyBtn and CopyBtn.Parent then
                CopyBtn.Text = "Copy Link"
            end
        end)
    else
        setStatus("Clipboard unsupported. Link: " .. Config.KeyLink, Theme.Error)
    end
end)

-- DRAGGING (mouse + touch)
---------------------------------------------------------------------
do
    local dragging, dragStart, startPos

    Main.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end
